    const defaultConfig = {
      chat_title: "Group Chat",
      welcome_message: "Welcome! Share your thoughts with everyone.",

      surface_color: "#ffffff",

    let messages = [];
    let messageCount = 0;
    let selectedPhoto = null;
    let userProfile = {
      displayName: ''
    };
    let lastSeenCount = 0;
    let originalTitle = 'Group Chat';

    const dataHandler = {
      onDataChanged(data) {
        messages = data.sort((a, b) => new Date(a.timestamp) - new Date(b.timestamp));
        messageCount = messages.length;
        renderMessages();
        checkMessageLimit();
        updateTabNotification();
      }
    };

    function updateTabNotification() {
      const unreadCount = messageCount - lastSeenCount;
      
      if (unreadCount > 0 && document.hidden) {
        document.title = `(${unreadCount}) ${originalTitle}`;
      } else {
        document.title = originalTitle;
        lastSeenCount = messageCount;
      }
    }

    function renderMessages() {
      const messagesArea = document.getElementById('messagesArea');
      const existingMessages = new Map(
        [...messagesArea.children].map(el => [el.dataset.timestamp, el])
      );

      messages.forEach(msg => {
        const key = msg.timestamp;
        if (existingMessages.has(key)) {
          existingMessages.delete(key);
        } else {
          const messageElement = createMessageElement(msg);
          messagesArea.appendChild(messageElement);
        }
      });

      existingMessages.forEach(el => el.remove());
      
      messagesArea.scrollTop = messagesArea.scrollHeight;
    }

    function createMessageElement(msg) {
      const bubble = document.createElement('div');
      bubble.className = 'message-bubble';
      bubble.dataset.timestamp = msg.timestamp;

      const username = document.createElement('div');
      username.className = 'message-username';
      username.textContent = msg.username;

      const text = document.createElement('div');
      text.className = 'message-text';
      
      // Check if message contains photo data
      if (msg.photoData) {
        const caption = document.createElement('div');
        caption.textContent = msg.message;
        caption.style.marginBottom = '8px';
        text.appendChild(caption);
        
        const img = document.createElement('img');
        img.src = msg.photoData;
        img.alt = msg.message;
        img.style.cssText = 'max-width: 100%; border-radius: 8px; margin-top: 8px; display: block;';
        text.appendChild(img);
      } else {
        text.textContent = msg.message;
      }

      const time = document.createElement('div');
      time.className = 'message-time';
      const date = new Date(msg.timestamp);
      time.textContent = date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      bubble.appendChild(username);
      bubble.appendChild(text);
      bubble.appendChild(time);

      return bubble;
    }

    function checkMessageLimit() {
      const limitWarning = document.getElementById('limitWarning');
      const sendButton = document.getElementById('sendButton');
      
      if (messageCount >= 999) {
        limitWarning.classList.add('show');
        sendButton.disabled = true;
      } else {
        limitWarning.classList.remove('show');
        sendButton.disabled = false;
      }
    }

    async function handleSubmit(event) {
      event.preventDefault();

      if (messageCount >= 999) {
        return;
      }

      const messageInput = document.getElementById('messageInput');
      const sendButton = document.getElementById('sendButton');

      const username = userProfile.displayName;
      const message = messageInput.value.trim();

      if (!username || !message) return;

      sendButton.disabled = true;
      sendButton.textContent = 'Sending...';

      const result = await window.dataSdk.create({
        username: username,
        message: message,
        photoData: null,
        timestamp: new Date().toISOString()
      });

      if (result.isOk) {
        messageInput.value = '';
        messageInput.focus();
      } else {
        showInlineError('Failed to send message. Please try again.');
      }

      sendButton.disabled = false;
      sendButton.textContent = 'Send';
    }

    function handlePhotoSelect(event) {
      const file = event.target.files[0];
      if (!file) return;

      if (!file.type.startsWith('image/')) {
        showInlineError('Please select a valid image file');
        return;
      }

      const reader = new FileReader();
      reader.onload = function(e) {
        // Compress the image before storing
        compressImage(e.target.result, file.name);
      };

      reader.readAsDataURL(file);
    }

    function compressImage(dataUrl, fileName) {
      const img = new Image();
      img.onload = function() {
        const canvas = document.createElement('canvas');
        const ctx = canvas.getContext('2d');
        
        // Resize to max 800px width while maintaining aspect ratio
        let width = img.width;
        let height = img.height;
        const maxWidth = 800;
        
        if (width > maxWidth) {
          height = (height * maxWidth) / width;
          width = maxWidth;
        }
        
        canvas.width = width;
        canvas.height = height;
        
        // Draw and compress
        ctx.drawImage(img, 0, 0, width, height);
        const compressedDataUrl = canvas.toDataURL('image/jpeg', 0.7);
        
        selectedPhoto = {
          name: fileName,
          dataUrl: compressedDataUrl
        };

        const photoPreview = document.getElementById('photoPreview');
        const photoStatus = document.getElementById('photoStatus');
        const photoStatusMessage = document.getElementById('photoStatusMessage');
        const sendPhotoButton = document.getElementById('sendPhotoButton');

        photoPreview.innerHTML = `
          <img src="${compressedDataUrl}" alt="Selected photo" style="max-width: 100%; max-height: 100%; object-fit: contain; border-radius: 8px;">
        `;

        photoStatus.textContent = '✓ Photo Selected';
        photoStatusMessage.textContent = `${fileName} - Ready to send to everyone`;
        sendPhotoButton.style.display = 'flex';
      };
      
      img.src = dataUrl;
    }

    async function sendPhoto() {
      if (!selectedPhoto || messageCount >= 999) return;

      const sendPhotoButton = document.getElementById('sendPhotoButton');
      sendPhotoButton.disabled = true;
      sendPhotoButton.textContent = '📤 Sending...';

      const username = userProfile.displayName;

      const result = await window.dataSdk.create({
        username: username,
        message: `📷 Photo: ${selectedPhoto.name}`,
        photoData: selectedPhoto.dataUrl,
        timestamp: new Date().toISOString()
      });

      if (result.isOk) {
        showInlineSuccess('Photo sent to everyone!');
        closePhotoModal();
      } else {
        showInlineError('Failed to send photo. Please try again.');
        sendPhotoButton.disabled = false;
        sendPhotoButton.textContent = '📤 Send to Everyone';
      }
    }

    }

          historyUsername.textContent = msg.username;

          const historyText = document.createElement('div');
          historyText.style.cssText = 'color: #e0e0e0; margin-bottom: 4px; font-size: 14px; line-height: 1.4;';
          
          if (msg.photoData) {
            const caption = document.createElement('div');
            caption.textContent = msg.message;
            caption.style.marginBottom = '8px';
            historyText.appendChild(caption);
            
            const img = document.createElement('img');
            img.src = msg.photoData;
            img.alt = msg.message;
            img.style.cssText = 'max-width: 100%; max-height: 200px; border-radius: 8px; margin-top: 8px; display: block;';
            historyText.appendChild(img);
          } else {
            historyText.textContent = msg.message;
          }

          const historyTime = document.createElement('div');
          historyTime.style.cssText = 'color: #999; font-size: 11px;';
          const date = new Date(msg.timestamp);
          historyTime.textContent = date.toLocaleString([], { 
            month: 'short', 
            day: 'numeric', 
            hour: '2-digit', 
            minute: '2-digit' 
          });

          historyItem.appendChild(historyUsername);
          historyItem.appendChild(historyText);
          historyItem.appendChild(historyTime);

          historyContent.appendChild(historyItem);
        });
      }

      historyModal.classList.add('active');
    }

    }

    function openProfile() {
      const profileModal = document.getElementById('profileModal');
      const displayNameInput = document.getElementById('displayNameInput');
      const profileError = document.getElementById('profileError');
      const profileSuccess = document.getElementById('profileSuccess');

      displayNameInput.value = userProfile.displayName;

      const profileModal = document.getElementById('profileModal');

    }

    function handleProfileSubmit(event) {
      event.preventDefault();

      const displayNameInput = document.getElementById('displayNameInput');
      const profileError = document.getElementById('profileError');
      const profileSuccess = document.getElementById('profileSuccess');

      const displayName = displayNameInput.value.trim();

      profileError.style.display = 'none';
      profileSuccess.style.display = 'none';

      if (!displayName) {
        profileError.textContent = 'Display name is required';
        profileError.style.display = 'block';
        return;
      }

      userProfile.displayName = displayName;

      profileSuccess.textContent = 'Profile updated successfully!';
      profileSuccess.style.display = 'block';

      setTimeout(() => {
        closeProfile();
      }, 1500);
    }
